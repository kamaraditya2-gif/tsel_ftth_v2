# ============================================================
# LANGKAH 1: Export dari server sumber (cachyos-x8664)
# ============================================================

# 1a. Dump database PostgreSQL
docker exec mojojojo_postgres pg_dump -U mojojojo_user -d mojojojo_database \
  --no-owner --no-privileges > /tmp/mojo_dump.sql
# Size: 2.7 MB

# 1b. Export Docker images
docker save mojo-worker:latest mojo-dashboard:latest -o /tmp/mojo_images.tar
# Size: 346 MB

# 1c. Export konfigurasi
tar -czf /tmp/mojo_configs.tar.gz \
  --exclude='data' --exclude='node_modules' \
  docker-compose*.yml .env .env.example schema.sql migrations/ config/ \
  nginx/ worker/ dashboard/Dockerfile dashboard/package.json \
  dashboard/postcss.config.js dashboard/tailwind.config.js \
  dashboard/next.config.js dashboard/tsconfig.json
# Size: 62 KB

# ============================================================
# LANGKAH 2: Transfer ke server tujuan
# ============================================================
sshpass -p '****' scp -P 6022 /tmp/mojo_dump.sql \
  /tmp/mojo_images.tar \
  /tmp/mojo_configs.tar.gz \
  webapp@gandooz.cloud:~/

# ============================================================
# LANGKAH 3: Di server tujuan - extract & load
# ============================================================
cd ~
tar -xzf mojo_configs.tar.gz
docker load -i mojo_images.tar

# ============================================================
# LANGKAH 4: Deploy dengan docker compose
# ============================================================
# ⚠️ PENTING: Set COMPOSE_PROJECT_NAME=mojojojomonitor agar:
#   - Worker terminal di dashboard bisa deteksi container
#   - Network naming konsisten dengan source
#   - Sesuai dengan label yang di-hardcode di worker-logs API

COMPOSE_PROJECT_NAME=mojojojomonitor docker compose up -d --no-build

# ============================================================
# TROUBLESHOOT 1: Build dashboard gagal
# ============================================================
# Error: Couldn't find any `pages` or `app` directory
# Penyebab: Source code dashboard tidak lengkap di server tujuan
#           (hanya Dockerfile, package.json, configs)
# Solusi: Gunakan --no-build agar pake pre-built image dari tar
#         docker compose up -d --no-build

# ============================================================
# TROUBLESHOOT 2: Worker Terminal di dashboard kosong
# ============================================================
# Gejala: Buka /admin/worker -> tidak ada log worker
# Penyebab: Container label project tidak cocok.
#           Worker-logs API hardcode filter:
#           -f label=com.docker.compose.project=mojojojomonitor
#           Jika project name != mojojojomonitor, 0 worker terdeteksi.
# Solusi: Set COMPOSE_PROJECT_NAME=mojojojomonitor saat docker compose up

# ============================================================
# TROUBLESHOOT 3: Regional worker tidak start
# ============================================================
# Gejala: network mojojojomonitor_mojojojo_network not found
# Penyebab: Network name di docker-compose.regional-r3-r12.yml
#           beda dengan yang dibuat compose (tergantung project name)
# Solusi: Sesuaikan network name di compose file regional:
#
#   sed -i 's|xxx_mojojojo_network|mojojojomonitor_mojojojo_network|' \
#     docker-compose.regional-r3-r12.yml
#
#   COMPOSE_PROJECT_NAME=mojojojomonitor docker compose \
#     -f docker-compose.regional-r3-r12.yml up -d

# ============================================================
# TROUBLESHOOT 4: Container name conflict
# ============================================================
# Gejala: Conflict. The container name "/xxx" is already in use
# Penyebab: Container dari percobaan sebelumnya masih ada
# Solusi: Bersihkan semua container + network lama dulu:
#
#   docker stop $(docker ps -q --filter 'name=mojo') \
#             $(docker ps -q --filter 'name=fping') 2>/dev/null
#   docker rm $(docker ps -aq --filter 'name=mojo') \
#            $(docker ps -aq --filter 'name=fping') 2>/dev/null
#   docker network prune -f

# ============================================================
# TROUBLESHOOT 5: Database hilang setelah postgres recreate
# ============================================================
# Gejala: Setelah restart compose, tabel kosong
# Penyebab: docker compose down + up me-recreate container
# Solusi: Import ulang dari dump:
#
#   docker exec -i mojojojo_postgres psql -U mojojojo_user \
#     -d mojojojo_database < ~/mojo_dump.sql

# ============================================================
# LANGKAH 5: Import database
# ============================================================
docker exec -i mojojojo_postgres psql -U mojojojo_user \
  -d mojojojo_database < mojo_dump.sql

# ============================================================

docker ps --format 'table {{.Names}}\t{{.Status}}' | grep -E 'mojo|fping'
